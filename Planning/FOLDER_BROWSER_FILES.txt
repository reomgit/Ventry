Usage Example
-------------

import SwiftUI

struct MyView: View {
    var body: some View {
        NavigationLink(destination: FolderBrowserView(
            title: "My Plugins",
            path: "/Applications/Adobe After Effects 2025/Plug-ins"
        )) {
            Text("Browse Plugins")
        }
    }
}
